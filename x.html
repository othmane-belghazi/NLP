- un dict {"statut": "ok"|"alerte"|"ko", ...} . affiché en POINT DE CONTRÔLE # - un dict {"labels": [...], "series": [...]} .. affiché en GRAPHIQUE Chart.js # - un dict {"colonnes": [...], "lignes": [...]} affiché en TABLE # # Dernière ligne : displayHTML(...) affiche le rapport interactif. # ============================================================================ import base64 import io import json from datetime import datetime # ---------------------------------------------------------------------------- # AIDES FACULTATIVES — à utiliser dans vos fonctions si vous voulez # préciser titre, statut, seuil, etc. Sinon la détection auto s'en charge. # ---------------------------------------------------------------------------- def checkpoint(libelle, statut, valeur, seuil="", commentaire=""): """statut : 'ok' | 'alerte' | 'ko'""" return {"_type": "checkpoint", "libelle": libelle, "statut": statut, "valeur": str(valeur), "seuil": str(seuil), "commentaire": commentaire} def table(titre, colonnes, lignes): return {"_type": "table", "titre": titre, "colonnes": list(colonnes), "lignes": [[str(v) for v in l] for l in lignes]} def visuel(titre, type_graph, labels, series): """type_graph : 'line' | 'bar' | 'doughnut' series : [{"nom": "...", "valeurs": [...]}, ...]""" return {"_type": "visuel", "titre": titre, "type": type_graph, "labels": list(labels), "series": series} # ---------------------------------------------------------------------------- # DÉTECTION AUTOMATIQUE de l'output d'une fonction — ne pas modifier # ---------------------------------------------------------------------------- _MAX_LIGNES_TABLE = 500 # sécurité d'affichage pour les DataFrames def _df_pandas_en_table(df, titre): df = df.head(_MAX_LIGNES_TABLE) return table(titre, [str(c) for c in df.columns], df.astype(str).values.tolist()) def _figure_en_image(fig, titre): buf = io.BytesIO() fig.savefig(buf, format="png", dpi=110, bbox_inches="tight") b64 = base64.b64encode(buf.getvalue()).decode("ascii") return {"_type": "visuel", "titre": titre, "type": "image", "contenu": "data:image/png;base64," + b64} def _normaliser(sortie, nom_fonction): """Transforme l'output brut d'une fonction en éléments affichables.""" if sortie is None: return [] if isinstance(sortie, (list, tuple)): elements = [] for s in sortie: elements.extend(_normaliser(s, nom_fonction)) return elements # Dict : déjà typé par les aides, ou reconnu par ses clés if isinstance(sortie, dict): d = dict(sortie) if "_type" not in d: if "statut" in d: d.setdefault("libelle", nom_fonction) d.setdefault("valeur", "") d.setdefault("seuil", "") d.setdefault("commentaire", "") d["_type"] = "checkpoint" elif "labels" in d and "series" in d: d.setdefault("titre", nom_fonction) d.setdefault("type", "bar") d["_type"] = "visuel" elif "colonnes" in d and "lignes" in d: d.setdefault("titre", nom_fonction) d["_type"] = "table" else: # dict simple {label: valeur} -> table à 2 colonnes return [table(nom_fonction, ["Indicateur", "Valeur"], [[k, v] for k, v in d.items()])] if d["_type"] == "checkpoint": d["valeur"] = str(d.get("valeur", "")) return [d] # DataFrame Spark if hasattr(sortie, "toPandas"): return [_df_pandas_en_table(sortie.limit(_MAX_LIGNES_TABLE).toPandas(), nom_fonction)] # DataFrame pandas if hasattr(sortie, "columns") and hasattr(sortie, "values"): return [_df_pandas_en_table(sortie, nom_fonction)] # Figure matplotlib if hasattr(sortie, "savefig"): return [_figure_en_image(sortie, nom_fonction)] # Axes matplotlib if hasattr(sortie, "get_figure"): return [_figure_en_image(sortie.get_figure(), nom_fonction)] # Tout le reste (str, nombre...) -> point d'information neutre return [checkpoint(libelle=nom_fonction, statut="ok", valeur=str(sortie))] # ---------------------------------------------------------------------------- # ASSEMBLAGE ET AFFICHAGE — ne pas modifier # ---------------------------------------------------------------------------- def generer_rapport(pipeline, periode=""): rapport = { "meta": {"periode": periode or datetime.now().strftime("%B %Y"), "genere_le": datetime.now().strftime("%Y-%m-%d %H:%M")}, "ordre_sections": list(pipeline.keys()), "sections": {}, } for cle, cfg in pipeline.items(): section = {"libelle": cfg.get("libelle", cle), "description": cfg.get("description", ""), "kpis": [{"label": k, "valeur": str(v)} for k, v in cfg.get("kpis", {}).items()], "checkpoints": [], "tables": [], "visuels": []} for fonction in cfg.get("fonctions", []): nom = getattr(fonction, "__name__", str(fonction)) try: elements = _normaliser(fonction(), nom) except Exception as e: # une fonction en erreur devient un KO visible elements = [checkpoint(libelle=f"Exécution de {nom}", statut="ko", valeur="ERREUR", commentaire=str(e))] for el in elements: el.setdefault("source_fonction", nom) genre = el.pop("_type") if genre == "checkpoint": el["fonction"] = el.pop("source_fonction") section["checkpoints"].append(el) elif genre == "table": section["tables"].append(el) else: section["visuels"].append(el) rapport["sections"][cle] = section return _GABARIT_HTML.replace("__DONNEES_JSON__", json.dumps(rapport, ensure_ascii=False)) #
